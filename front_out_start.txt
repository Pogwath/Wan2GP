 185:         """
 186:         将 jqx 的韵母为 u/ü 的拼音转换为 v
 187:         如：ju -> jv , que -> qve, xün -> xvn
 188:         """

 190:             return pinyin

 194:         pinyin = re.sub(pattern, repl, pinyin, flags=re.IGNORECASE)

 231:     def save_tech_terms(self, original_text):
 232:         """
 233:         保护技术术语中的连字符，防止被中文normalizer解析为减号
 234:         策略：将术语中的连字符替换为特殊占位符<H>，数字仍可被正常处理
 235:         例如：GPT-5-nano -> GPT<H>5<H>nano，然后 5 被转换为 五
 236:         最终恢复为：GPT-五-nano
 237:         """
 238:         tech_pattern = re.compile(TextNormalizer.TECH_TERM_PATTERN)
 239:         original_tech_list = tech_pattern.findall(original_text)
 240:         if len(original_tech_list) == 0:
 241:             return (original_text, None)
 242:
 243:         # 去重并按长度降序排列（避免短匹配先替换导致问题）
 244:         original_tech_list = sorted(set(original_tech_list), key=len, reverse=True)
 245:         transformed_text = original_text
 246:
 247:         # 将术语中的连字符替换为占位符 <H>
 248:         for term in original_tech_list:
 249:             # 将 GPT-5-nano 替换为 GPT<H>5<H>nano
 250:             protected_term = term.replace("-", "<H>")
 251:             transformed_text = transformed_text.replace(term, protected_term)
 252:
 253:         return transformed_text, original_tech_list

 255:     def restore_tech_terms(self, normalized_text, original_tech_list):
 256:         """
 257:         恢复技术术语中的连字符
 258:         将占位符 <H> 恢复为连字符 -
 259:         同时清理 normalizer 可能在占位符周围添加的多余空格
 260:         """
 261:         if not original_tech_list or len(original_tech_list) == 0:
 262:             return normalized_text
 263:
 264:         # 清理 <H> 周围可能的空格，然后恢复为连字符
 265:         # 处理模式: " <H> " -> "-", " <H>" -> "-", "<H> " -> "-", "<H>" -> "-"
 266:         transformed_text = re.sub(r'\s*<H>\s*', '-', normalized_text)
 267:         return transformed_text

 269:     def apply_glossary_terms(self, text, lang="zh"):
 270:         """
 271:         应用术语词汇表，将专业术语替换为对应语言的读法
 272:
 273:         Args:
 274:             text: 待处理文本
 275:             lang: 语言类型 "zh" 或 "en"
 276:
 277:         Returns:
 278:             处理后的文本
 279:
 280:         Example:
 281:             "M.2 NVMe SSD" -> (zh) "M 二 NVMe SSD"
 282:             "M.2 NVMe SSD" -> (en) "M dot two NVMe SSD"
 283:         """
 284:         if not self.term_glossary:
 285:             return text
 286:
 287:         # 按术语长度降序排列，避免短术语先匹配导致长术语无法匹配
 288:         # 例如："PCIe 5.0" 应该在 "PCIe" 之前匹配
 289:         sorted_terms = sorted(self.term_glossary.keys(), key=len, reverse=True)
 290:         @lru_cache(maxsize=42)
 291:         def get_term_pattern(term: str):
 292:             return re.compile(re.escape(term), re.IGNORECASE)
 293:         transformed_text = text
 294:         for term in sorted_terms:
 295:             term_value = self.term_glossary[term]
 296:             if isinstance(term_value, dict):
 297:                 replacement = term_value.get(lang, term_value.get(lang, term))
 298:             else:
 299:                 replacement = term_value
 300:             # 使用正则进行大小写不敏感的替换
 301:             pattern = get_term_pattern(term)
 302:             transformed_text = pattern.sub(replacement, transformed_text)
 303:
 304:         return transformed_text
